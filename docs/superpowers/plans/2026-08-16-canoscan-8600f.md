# CanoScan 8600F Driver Stack + scan8600 CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Selbst kompilierter SANE-Treiberstack (genesys) plus Python-CLI `scan8600` für Flachbett- und Durchlicht-Scans mit dem Canon CanoScan 8600F auf macOS arm64.

**Architecture:** sane-backends wird aus dem Quellcode gebaut und repo-lokal unter `./prefix` installiert (nur genesys-Backend). Die CLI ruft das lokale `scanimage` per subprocess, TIFF nativ, PNG/JPEG via Pillow.

**Tech Stack:** autotools, libusb (Homebrew), sane-backends 1.3.1, Python ≥3.13, Pillow, pytest.

**Spec:** `docs/superpowers/specs/2026-08-16-canoscan-8600f-design.md`

## Global Constraints

- Kein System-Install: alles unter `~/Documents/GitHub/canoscan` bzw. `./prefix`.
- Nur genesys-Backend (`BACKENDS=genesys`).
- Python nur stdlib + Pillow (Tests: pytest).
- Default-Format TIFF; Defaults 300 dpi (flatbed) / 1200 dpi (film).
- Hardware-Tests brauchen den angeschlossenen 8600F (USB 04a9:2229).

```python
#!/usr/bin/env python3
"""scan8600 - Flachbett- und Durchlicht-Scans mit dem CanoScan 8600F."""
import argparse
import datetime
import pathlib
import re
import subprocess
import sys

PREFIX = pathlib.Path(__file__).resolve().parent / "prefix"
SCANIMAGE = PREFIX / "bin" / "scanimage"
SANE_ENV = {"SANE_CONFIG_DIR": str(PREFIX / "etc" / "sane.d")}
DEFAULT_DPI = {"flatbed": 300, "film": 1200}


def parse_args(argv):
    p = argparse.ArgumentParser(prog="scan8600")
    p.add_argument("--mode", required=True, choices=["flatbed", "film"])
    p.add_argument("--dpi", type=int)
    p.add_argument("--format", choices=["tiff", "png", "jpeg"], default="tiff")
    p.add_argument("--output")
    p.add_argument("--gray", action="store_true")
    a = p.parse_args(argv)
    if a.dpi is None:
        a.dpi = DEFAULT_DPI[a.mode]
    return a


def default_output(a):
    ext = {"tiff": "tiff", "png": "png", "jpeg": "jpg"}[a.format]
    stamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
    return f"scan_{stamp}.{ext}"


def build_command(a, source_name):
    cmd = [str(SCANIMAGE), "--format=tiff",
           "--resolution", str(a.dpi),
           "--mode", "Gray" if a.gray else "Color"]
    if source_name:
        cmd += ["--source", source_name]
    return cmd
```

```python
class ScanError(Exception):
    pass


def find_film_source(options_text):
    m = re.search(r"--source\s+(\S[^\[\n]*)", options_text)
    if not m:
        return None
    for cand in m.group(1).split("|"):
        cand = cand.strip()
        if re.search(r"transparen|film|ta\b", cand, re.IGNORECASE):
            return cand
    return None


def scanimage_run(cmd, **kw):
    import os
    env = dict(os.environ, **SANE_ENV)
    return subprocess.run(cmd, env=env, capture_output=True, **kw)


def run_scan(a):
    out = pathlib.Path(a.output or default_output(a))
    source = None
    if a.mode == "film":
        probe = scanimage_run([str(SCANIMAGE), "-A"])
        source = find_film_source(probe.stdout.decode(errors="replace"))
        if source is None:
            raise ScanError(
                "Keine Durchlicht-Quelle gefunden. Verfügbare Optionen:\n"
                + probe.stdout.decode(errors="replace"))
    r = scanimage_run(build_command(a, source))
    if r.returncode != 0:
        err = r.stderr.decode(errors="replace")
        if "no SANE devices" in err or "Invalid argument" in err:
            raise ScanError(
                "Scanner nicht gefunden. USB-Kabel prüfen, dann: "
                "ioreg -p IOUSB | grep -i CanoScan. Details:\n" + err)
        raise ScanError(err)
    tiff_bytes = r.stdout
    if a.format == "tiff":
        out.write_bytes(tiff_bytes)
    else:
        import io
        from PIL import Image
        img = Image.open(io.BytesIO(tiff_bytes))
        img.save(out, quality=95)
    return out


def main(argv=None):
    a = parse_args(argv if argv is not None else sys.argv[1:])
    try:
        out = run_scan(a)
    except ScanError as e:
        print(f"scan8600: {e}", file=sys.stderr)
        return 1
    print(out)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

Hinweis: `run_scan` ruft intern `scanimage_run`; der Test patcht `scan8600.subprocess.run`, was durch `capture_output=True` weiter funktioniert — fake_run akzeptiert `**kw`.

```python
# descratch.py
"""IR-basierte Kratzer-/Staubentfernung (Prinzip SilverFast iSRD/FARE)."""
import cv2
import numpy as np


def defect_mask(ir_gray):
    thr, _ = cv2.threshold(ir_gray, 0, 255,
                           cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
    _, mask = cv2.threshold(ir_gray, thr, 255, cv2.THRESH_BINARY_INV)
    kernel = np.ones((5, 5), np.uint8)
    return cv2.dilate(mask, kernel, iterations=1)


def remove_defects(visible_rgb, ir_gray):
    if ir_gray.shape[:2] != visible_rgb.shape[:2]:
        ir_gray = cv2.resize(ir_gray,
                             (visible_rgb.shape[1], visible_rgb.shape[0]))
    mask = defect_mask(ir_gray)
    return cv2.inpaint(visible_rgb, mask, 3, cv2.INPAINT_TELEA)
```
